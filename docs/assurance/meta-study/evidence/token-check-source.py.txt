import hashlib,json,os
from pathlib import Path
os.environ['HF_HUB_OFFLINE']='1'
os.environ['TRANSFORMERS_OFFLINE']='1'
os.environ['HF_HUB_DISABLE_TELEMETRY']='1'
from transformers import AutoTokenizer
model='/workspace/models/meta-llama/Llama-3.2-1B-Instruct/9213176726f574b556790deb65791e0c5aa438b6'
path=Path('/workspace/simurgh-research/meta-study/capture.json')
capture=json.loads(path.read_text())
tokenizer=AutoTokenizer.from_pretrained(model,local_files_only=True,trust_remote_code=False)
checked=0
for row in capture['rows']:
 for step in row['steps']:
  if step['status']!='generated': continue
  ids=tokenizer.apply_chat_template(step['messages'],add_generation_prompt=True,date_string='15 Oct 2026',return_dict=True)['input_ids']
  assert ids==step['input_token_ids'],(row['id'],'input token mismatch')
  assert tokenizer.decode(ids,skip_special_tokens=False)==step['rendered_prompt'],row['id']
  assert tokenizer.decode(step['generated_token_ids'],skip_special_tokens=True)==step['completion'],row['id']
  checked+=1
print(json.dumps({'capture_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'generated_rows_checked':checked,
 'checks':['prompt to input token IDs','input IDs to rendered prompt','output IDs to completion'],
 'scope':'same installed local tokenizer; consistency check, not independent generation attestation'}))
